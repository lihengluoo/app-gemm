foreach(cpu RANGE 0 2)
    list(GET gemm_cpu_names ${cpu} cpu_name)
    add_executable(gemm_${cpu_name}_check gemm_check.c)
    target_link_libraries(gemm_${cpu_name}_check PRIVATE gemm_${cpu_name})
    target_compile_definitions(gemm_${cpu_name}_check PRIVATE gemm_cpu=${cpu})
    add_test(NAME gemm_${cpu_name}_check COMMAND gemm_${cpu_name}_check)
    set_tests_properties(gemm_${cpu_name}_check PROPERTIES TIMEOUT 180)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(gemm_${cpu_name}_fault_check gemm_fault_check.c)
        target_link_libraries(gemm_${cpu_name}_fault_check PRIVATE gemm_${cpu_name})
        target_link_options(gemm_${cpu_name}_fault_check PRIVATE
            -Wl,--wrap=pthread_create -Wl,--wrap=pthread_join -Wl,--wrap=aligned_alloc)
        add_test(NAME gemm_${cpu_name}_fault_check COMMAND gemm_${cpu_name}_fault_check)
        set_tests_properties(gemm_${cpu_name}_fault_check PROPERTIES TIMEOUT 30)
    endif()
endforeach()
