cmake_minimum_required(VERSION 3.16)
project(gemm LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
include(CTest)
add_compile_definitions(_POSIX_C_SOURCE=200809L)
add_compile_options(-Wall -Wextra -Werror -Wpedantic -Wstrict-prototypes)

set(gemm_common_files lib/gemm.c lib/gemm_pack.c lib/gemm_threads.c)
set(gemm_cpu_names scalar 6780e_avx 4509y_avx)
set(gemm_kernel_files lib/gemm_scalar.c lib/gemm_6780e.c lib/gemm_4509y.c)
set(gemm_backend scalar CACHE STRING "Kernel used by gemm_benchmark")
set_property(CACHE gemm_backend PROPERTY STRINGS ${gemm_cpu_names})
if(NOT gemm_backend IN_LIST gemm_cpu_names)
    message(FATAL_ERROR "Unknown GEMM backend: ${gemm_backend}")
endif()

foreach(cpu RANGE 0 2)
    list(GET gemm_cpu_names ${cpu} cpu_name)
    list(GET gemm_kernel_files ${cpu} kernel_file)
    add_library(gemm_${cpu_name} STATIC ${gemm_common_files} ${kernel_file})
    target_include_directories(gemm_${cpu_name} PUBLIC include)
    target_compile_definitions(gemm_${cpu_name} PRIVATE gemm_cpu=${cpu})
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
        target_compile_options(gemm_${cpu_name} PRIVATE -fno-vectorize -fno-slp-vectorize)
    elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU")
        target_compile_options(gemm_${cpu_name} PRIVATE -fno-tree-vectorize)
    endif()
    target_link_libraries(gemm_${cpu_name} PUBLIC Threads::Threads)
endforeach()

add_executable(gemm_benchmark gemm_benchmark.c)
target_link_libraries(gemm_benchmark PRIVATE gemm_${gemm_backend})
if(BUILD_TESTING)
    add_test(NAME gemm_benchmark COMMAND gemm_benchmark)
    set_tests_properties(gemm_benchmark PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    add_subdirectory(tests)
endif()
